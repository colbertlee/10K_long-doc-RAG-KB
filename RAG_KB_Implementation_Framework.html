项目目标与使用说明 Objectives & How to Use

本文档是 RAG_KB_Plan.html 的配套实施蓝图（Implementation Blueprint），在 不修改原规划文件 的前提下，将设计文档中的分层管道、技术选型、模块职责、开发阶段与验证清单转化为可执行的工程路线图。

本文件假设用户已确认技术栈：MVP 优先、Python 3.11 + pip、LightRAG 本地向量+图索引（NetworkX + NanoVectorDB + JSON）、Open WebUI 前端、Ollama 本地模型、原生 Windows；LightRAG 作为主力 RAG 引擎。

核心目标 Objectives

  • 将总体设计拆分为 8 个可验收的工程阶段（Phase 0–8）。
  • 提供接近可运行的接口签名、配置示例、目录结构与关键伪代码。
  • 锁定以 Python 3.11 + pip、LightRAG、Open WebUI、FastAPI 为核心的落地栈。
  • LightRAG 作为主力 RAG 引擎，同时保留自定义后端的扩展能力（source/category 子库、可选自定义 Embedding/LLM 注入）。
  • 确保权限（RBAC/ACL）、增量更新、可观测性内建于每个阶段。
  • 每个版本完成后推送 GitHub Release，附带 README、Release Notes 与一键安装说明，提升交付与可复现性。

如何使用 How to Use

  1. 按 Phase 0 → Phase 8 顺序推进，每阶段先跑通接口再填充实现细节。
  2. 代码块为骨架（Skeleton），需补充错误处理、日志、监控与集成测试。
  3. 使用 pytest 对每个阶段做验收，再进入下一阶段。
  4. 批量 10K 文档入库走 /api/v1/ingest 后端；Open WebUI 仅作为聊天前端入口。
  5. 在 configs/config.example.yaml 中切换 Ollama 与 OpenAI-compatible 远程模型。
  6. 每阶段完成后保持 README 与 Release Notes 同步；版本发布前使用 GitHub Release 归档可运行版本。

注：所有代码仅作为文档内示例，不是可直接上线的完整系统；生产环境需完善安全、并发、容错与审计。为降低用户门槛，建议提供 pip install -e . 或一键安装脚本，并在 README 中给出从安装到运行的完整步骤。

from __future__ import annotations
from datetime import datetime
from typing import Any, Dict, List, Optional
from pydantic import BaseModel, Field

class Document(BaseModel):
    doc_id: str = Field(..., description='Unique document id, e.g. sha256 of normalized content or file hash')
    title: str = ''
    source: str = ''
    content: str
    metadata: Dict[str, Any] = Field(default_factory=dict)
    file_hash: str = ''
    created_at: datetime = Field(default_factory=datetime.utcnow)
    updated_at: datetime = Field(default_factory=datetime.utcnow)
    acl: Dict[str, List[str]] = Field(default_factory=dict, description='RBAC/ACL tags, e.g. {"dept": ["Sales"], "level": ["Internal"]}')

class Chunk(BaseModel):
    chunk_id: str
    doc_id: str
    parent_id: Optional[str] = None
    text: str
    level: int = 0
    section_path: List[str] = Field(default_factory=list)
    token_count: int = 0
    metadata: Dict[str, Any] = Field(default_factory=dict)
    embedding: Optional[List[float]] = None

class SearchResult(BaseModel):
    chunk_id: str
    doc_id: str
    text: str
    score: float
    rank: int
    source: str = ''
    metadata: Dict[str, Any] = Field(default_factory=dict)

from rag_kb.models import Document

class BaseParser(ABC):
    supported_ext: tuple[str, ...] = ()

    @abstractmethod
    def parse(self, path: Path) -> Document:
        '''Return a Document with full text and metadata.'''
        ...

    def can_parse(self, path: Path) -> bool:
        return path.suffix.lower() in self.supported_ext

PyMuPDF 解析示例

# src/rag_kb/parsers/pdf_pymupdf.py
import hashlib
from pathlib import Path
import fitz  # PyMuPDF
from rag_kb.models import Document

class PyMuPDFParser:
    supported_ext = ('.pdf',)

    def parse(self, path: Path) -> Document:
        doc = fitz.open(path)
        parts = []
        for page in doc:
            parts.append(page.get_text())
        content = '\n\n'.join(parts)
        file_hash = hashlib.sha256(content.encode('utf-8')).hexdigest()
        return Document(
            doc_id=file_hash[:16],
            title=path.stem,
            source=str(path),
            content=content,
            metadata={'pages': len(doc), 'parser': 'pymupdf'},
            file_hash=file_hash,
        )

pdfplumber 解析示例（保留表格）

# src/rag_kb/parsers/pdf_pdfplumber.py
import hashlib
from pathlib import Path
import pdfplumber
from rag_kb.models import Document

class PDFPlumberParser:
    supported_ext = ('.pdf',)

    def parse(self, path: Path) -> Document:
        parts = []
        tables_md = []
        with pdfplumber.open(path) as pdf:
            for page in pdf.pages:
                text = page.extract_text() or ''
                parts.append(text)
                for table in (page.extract_tables() or []):
                    rows = ['| ' + ' | '.join(str(c or '') for c in row) + ' |' for row in table]
                    tables_md.append('\n'.join(rows))
        content = '\n\n'.join(parts)
        if tables_md:
            content += '\n\n## Tables\n' + '\n\n'.join(tables_md)
        file_hash = hashlib.sha256(content.encode('utf-8')).hexdigest()
        return Document(
            doc_id=file_hash[:16],
            title=path.stem,
            source=str(path),
            content=content,
            metadata={'pages': len(pdf.pages), 'parser': 'pdfplumber', 'tables': len(tables_md)},
            file_hash=file_hash,
        )

去重与 PII 脱敏占位

# src/rag_kb/ingest/cleaner.py
import re
from typing import List
from rag_kb.models import Document

def dedupe_by_hash(documents: List[Document]) -> List[Document]:
    seen = set()
    unique = []
    for doc in documents:
        if doc.file_hash in seen:
            continue
        seen.add(doc.file_hash)
        unique.append(doc)
    return unique

def mask_pii_placeholder(text: str) -> str:
    '''Placeholder PII de-identification: use regex for demo only.
    Replace with a dedicated NER/PII library in production.'''
    text = re.sub(r'\d{18}', '[ID]', text)  # Chinese ID
    text = re.sub(r'1[3-9]\d{9}', '[MOBILE]', text)
    text = re.sub(r'\d{4}[- ]?\d{4}[- ]?\d{4}[- ]?\d{4}', '[CARD]', text)
    return text

元数据绑定 Ingest Pipeline

# src/rag_kb/ingest/pipeline.py
from pathlib import Path
from typing import Optional
from rag_kb.models import Document
from rag_kb.ingest.cleaner import mask_pii_placeholder
from rag_kb.parsers.registry import PARSER_REGISTRY

class IngestPipeline:
    def run(self, file: Path, acl: Optional[dict] = None) -> Document:
        parser = next((p for p in PARSER_REGISTRY if p.can_parse(file)), None)
        if parser is None:
            raise ValueError(f'No parser registered for {file.suffix}')
        doc = parser.parse(file)
        if acl:
            doc.acl = acl
            for k, v in acl.items():
                # Keep source/category/ACL metadata scalar for downstream filtering and text header injection
                doc.metadata[f'acl_{k}'] = v[0] if isinstance(v, list) and v else str(v)
        doc.content = mask_pii_placeholder(doc.content)
        return doc

StructuredChunker：按标题层级切片

# src/rag_kb/chunkers/structured.py
import re
import uuid
from typing import List
from rag_kb.models import Document, Chunk
from rag_kb.chunkers.base import BaseChunker

class StructuredChunker(BaseChunker):
    '''Structure-aware chunker: split by H1..H6, merge small paragraphs,
    keep overlapping windows between sections.'''
    def __init__(self, target_tokens: int = 400, overlap_chars: int = 60):
        self.target_tokens = target_tokens
        self.overlap_chars = overlap_chars

    def _heading_level(self, line: str) -> int:
        m = re.match(r'^(#{1,6})\s+', line)
        return len(m.group(1)) if m else 0

    def chunk(self, doc: Document) -> List[Chunk]:
        chunks: List[Chunk] = []
        section_path: List[str] = []
        buffer = ''
        current_title = ''
        current_level = 0

        for line in doc.content.splitlines():
            level = self._heading_level(line)
            if level:
                if buffer.strip():
                    chunks.append(self._make_chunk(doc, buffer, current_title, section_path, current_level))
                    buffer = buffer[-self.overlap_chars:] if self.overlap_chars else ''
                current_title = line.strip().lstrip('#').strip()
                section_path = section_path[:level - 1] + [current_title]
                current_level = level
            else:
                buffer += line + '\n'

        if buffer.strip():
            chunks.append(self._make_chunk(doc, buffer, current_title, section_path, current_level))
        return chunks

    def _make_chunk(self, doc, text, title, path, level):
        return Chunk(
            chunk_id=str(uuid.uuid4()),
            doc_id=doc.doc_id,
            text=text.strip(),
            level=level,
            section_path=path,
            token_count=len(text) // 4,
            metadata={**doc.metadata, 'title': title},
        )

ParentChildChunker：小 Child 命中、大 Parent 补全上下文

# src/rag_kb/chunkers/parent_child.py
import uuid
from typing import List
from rag_kb.models import Document, Chunk
from rag_kb.chunkers.structured import StructuredChunker

class ParentChildChunker(StructuredChunker):
    '''Small child chunks for high-precision retrieval; parent chunk
    carries full section context for prompt expansion.'''
    def __init__(self, parent_target: int = 1200, child_target: int = 250, overlap_chars: int = 40):
        super().__init__(target_tokens=child_target, overlap_chars=overlap_chars)
        self.parent_target = parent_target

    def chunk(self, doc: Document) -> List[Chunk]:
        sections = super().chunk(doc)
        children: List[Chunk] = []
        for parent in sections:
            parent.chunk_id = str(uuid.uuid4())
            parent.token_count = len(parent.text) // 4
            words = parent.text.split()
            start = 0
            while start < len(words):
                end = min(start + self.target_tokens, len(words))
                child_text = ' '.join(words[start:end])
                children.append(Chunk(
                    chunk_id=str(uuid.uuid4()),
                    doc_id=doc.doc_id,
                    parent_id=parent.chunk_id,
                    text=child_text,
                    level=parent.level,
                    section_path=parent.section_path,
                    token_count=len(child_text) // 4,
                    metadata={'parent_text': parent.text, **doc.metadata},
                ))
                start = end
        return children

  2. 相邻 chunk 保留 40–60 字符重叠，避免句子断裂。
  3. Child chunk 通过 parent_id 可回溯到 Parent chunk 全文。
  4. 表格内容不会被切成半截行。

# src/rag_kb/lightrag/adapter.py
import asyncio
import json
from pathlib import Path
from lightrag import LightRAG, QueryParam
from rag_kb.lightrag.llm_funcs import ollama_llm
from rag_kb.lightrag.embedding_funcs import ollama_embed
from rag_kb.config import settings

NL = chr(10)

class LightRAGAdapter:
    def __init__(self, working_dir=None):
        self.working_dir = Path(working_dir or settings.lightrag_working_dir)
        self.working_dir.mkdir(parents=True, exist_ok=True)
        self.rag = LightRAG(
            working_dir=str(self.working_dir),
            llm_model_func=ollama_llm,
            embedding_func=ollama_embed,
            chunk_token_size=settings.lightrag_chunk_token_size,
            max_token=settings.lightrag_max_token,
        )

    def insert_chunks(self, chunks):
        parts = []
        for c in chunks:
            meta = getattr(c, 'metadata', {}) or {}
            header = (
                '[source=' + str(meta.get('source', '')) +
                ';category=' + str(meta.get('category', '')) +
                ';product_id=' + str(meta.get('product_id', '')) +
                ';doc_id=' + str(c.doc_id) + ']'
            )
            parts.append(header + NL + c.text)
        doc_text = (NL + NL).join(parts)
        self.rag.insert(doc_text)

    def query(self, question, mode=None):
        mode = mode or settings.lightrag_query_mode
        return self.rag.query(
            question,
            param=QueryParam(mode=mode, only_need_context=False),
        )

    async def stream_query(self, question, mode=None):
        loop = asyncio.get_event_loop()
        mode = mode or settings.lightrag_query_mode
        answer = await loop.run_in_executor(
            None,
            self.rag.query,
            question,
            QueryParam(mode=mode, only_need_context=False),
        )
        SSE_END = NL * 2
        buf = ''
        for ch in answer:
            buf += ch
            if ch in ('。', '？', '！', '.', '?', '!', NL):
                payload = json.dumps({'choices': [{'delta': {'content': buf}}]})
                yield 'data: ' + payload + SSE_END
                buf = ''
        if buf:
            yield 'data: ' + json.dumps({'choices': [{'delta': {'content': buf}}]}) + SSE_END
        yield 'data: [DONE]' + SSE_END

Phase 4 LightRAG 多模式查询与生成 LightRAG Query Modes & Generation

LightRAG 提供 naive、local、global、hybrid 四种查询模式。推荐默认使用 hybrid，既能利用向量相似，也能通过图扩展召回相关上下文。

  • naive：仅向量检索，适合简单事实查询。
  • local：基于局部图子结构与向量，适合需要实体关系的问题。
  • global：基于全局图摘要，适合宏观/聚合问题。
  • hybrid：综合局部与全局，兼顾精度与覆盖，默认。

查询模式示例

按 source/category/product_id 过滤

LightRAG 目前不直接支持 metadata 过滤。推荐两种策略：

  • 预过滤 Pre-filtering：入库时按 category/product 拆分为多个 LightRAG 工作目录（子库），查询时先选择对应 adapter。
  • 后过滤 Post-filtering：全库 hybrid 查询后，对返回的 chunk/source 元数据做正则/字典匹配，剔除不在目标范围的片段；必要时重新查询。

from rag_kb.models import SearchResult

def simple_faithfulness(answer: str, contexts: List[SearchResult]) -> float:
    '''Simplified faithfulness: ratio of answer words present in contexts.'''
    answer_words = set(answer.lower().split())
    context_words = set(' '.join(c.text.lower() for c in contexts).split())
    if not answer_words:
        return 0.0
    return len(answer_words & context_words) / len(answer_words)

文件 Hash 增量更新与映射

# src/rag_kb/ingest/incremental.py
import hashlib
import json
import shutil
from pathlib import Path
from typing import List, Set, Tuple

CHUNK_MAP = Path('./data/chunk_map.json')
CATEGORY_DIR = Path('./data/category_dbs')

def load_doc_chunk_map():
    if CHUNK_MAP.exists():
        return json.loads(CHUNK_MAP.read_text(encoding='utf-8'))
    return {}

def save_doc_chunk_map(m: dict):
    CHUNK_MAP.write_text(json.dumps(m, indent=2, ensure_ascii=False), encoding='utf-8')

def compute_file_hash(path: Path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

def plan_update(docs_dir: Path, known: dict):
    current = {}
    for p in docs_dir.rglob('*'):
        if p.is_file():
            current[compute_file_hash(p)] = p
    to_add = [v for h, v in current.items() if h not in known]
    to_delete = {doc_id for doc_id, info in known.items() if info.get('hash') not in current}
    return to_add, to_delete

def update_doc_chunk_map(doc_id: str, file_hash: str, source: str, category: str, m: dict):
    m[doc_id] = {'hash': file_hash, 'source': source, 'category': category}

LightRAG 索引更新策略

LightRAG 当前不直接支持按 doc_id 删除单个 chunk。推荐按 source/category 拆分子库，更新时重建对应子库，避免脏索引。

# src/rag_kb/ingest/incremental.py (continued)
def rebuild_category(category: str, docs: list, adapter_factory):
    cat_dir = CATEGORY_DIR / category
    if cat_dir.exists():
        shutil.rmtree(cat_dir)
    adapter = adapter_factory(cat_dir)
    for doc in docs:
        adapter.insert_chunks(doc.chunks)
    # Note: caller updates chunk_map with current docs after rebuild
    return adapter

# On delete/update:
# 1. Load chunk_map, find affected doc_ids and their categories.
# 2. Collect remaining docs in each affected category.
# 3. rebuild_category(category, docs, adapter_factory)
# 4. Update chunk_map.

风险与常见陷阱 Risks & Pitfalls

以下为落地过程中最常见的风险与缓解措施。

风险矩阵 Risk Matrix

风险 Risk影响 Impact缓解 Mitigation
Python 3.11 环境未就绪Open WebUI 无法安装部署机器使用 Python 3.11；开发机可用 miniconda 创建独立环境
解析库体积大 / 安装失败pymupdf / pdfplumber 在 Windows 下编译报错使用预编译 wheel；必要时降级到稳定版本
Windows 多进程 / 路径问题LightRAG / uvicorn 启动异常避免路径中的空格；使用 if __name__ == "__main__": 保护
Ollama 单点性能不足并发 embedding/LLM 延迟高开启 Ollama GPU 支持；远程模型做 fallback
LightRAG 本地存储写冲突多 worker 同时写入导致索引损坏索引服务单写者多读者；队列化批量入库
Open WebUI 被当作完整 RAG复杂切片、图增强检索、增量更新缺失明确前后端边界，Open WebUI 只做聊天界面
LightRAG 抽取噪声错误关系被放大规则后处理 + 抽样人工校验 + 分阶段上线
ACL 未在检索层过滤越权访问敏感文档将 ACL 作为 source/category 过滤条件与后端二次过滤
常见陷阱 Common Pitfalls
  • Lost in the Middle：LLM 对长上下文中间信息关注度下降，送入 LLM 的 chunk 控制在 3–8 个。
  • 表格直接切片会丢失行列关系；应转为 Markdown 并在命中后传原表。
  • 缺少 DocID↔ChunkID 映射会导致更新/删除后向量库残留过期数据。
  • 仅依赖单一查询模式会漏掉关系或事实；应根据问题类型切换 hybrid/local/global/naive。
  • 没有 RAGAS 或类似评估，无法量化优化效果。
